# fly-ticket — server crontab (install: crontab /path/to/fly-ticket/crontab.txt)
# All jobs point at src/jobs.php; logs go to storage/logs/ (see app-YYYY-MM-DD.log).

# cron does not read .bashrc, so `php` would resolve to the system build, which
# on this machine lacks pdo_pgsql. Put herd-lite's PHP first on PATH (its ini is
# scanned from the binary's own directory). Harmless elsewhere; drop this line
# if the server's system PHP already has pdo_pgsql.
PATH=/home/kinder/.config/herd-lite/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# Fetch prices 4×/day: 06:00, 11:00, 16:00, 20:00
0 6,11,16,20 * * * cd /home/kinder/Documents/fly-ticket && php src/jobs.php fetch >> storage/logs/cron-$(date +\%F).log 2>&1

# Daily price report (fan-out to every configured bot) at 08:00.
# Each bot reports to its own chat — see runReportJob() in src/jobs.php.
0 8 * * * cd /home/kinder/Documents/fly-ticket && php src/jobs.php report >> storage/logs/cron-$(date +\%F).log 2>&1

# Keep the bots alive: restart if not running (every minute).
# `php src/jobs.php bot` spawns one poller per configured bot and waits on all,
# so a single watchdog line covers every bot. The pgrep pattern matches each
# child poller ("php ... telegram-bot.php <name>"). NOTE: the pattern uses [.]
# (regex for a literal dot) so it matches the real process but does NOT self-match
# this crontab line — a bare "telegram-bot.php" would make pgrep match cron's own
# shell and the bot would never start.
* * * * * pgrep -f "telegram-bot[.]php" > /dev/null || (cd /home/kinder/Documents/fly-ticket && nohup php src/jobs.php bot >> storage/logs/bot-$(date +\%F).log 2>&1 &)

# Log retention: remove log files older than 30 days. Date-stamped files only
# get written on their own day, so mtime +30 is when they really go stale —
# this covers app-*, cron-* and bot-* logs.
0 3 * * * find /home/kinder/Documents/fly-ticket/storage/logs -name "*.log" -mtime +30 -delete >> /dev/null 2>&1
